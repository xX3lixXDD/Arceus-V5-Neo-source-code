--[[ Enums ]]

do
	local enum = {}
	local enums = {}

	function enum.__index(t, k)
		return t._map[k] or enums[k]
	end

	function enum.new(...: any): {any}
		local items = {...}
		local map = {}

		for i, v in ipairs(items) do
			map[v] = i
		end

		return setmetatable({
			_map = map,
			_items = items
		}, enum)
	end

	function enum:GetEnumItems()
		return self._items
	end

	function enums:AddEnum(category: string, ...: string)
		enums[category] = enum.new(...)
	end

	wave.enums = enums;
end

	end

				if not dest[name] then
					local tbl = {}
					env.protectfunction(tbl)
					dest[name] = tbl
				end

				recursive(dest[name], closure)
			elseif typeof(closure) == "function" then
				dest[name] = env.GCProtectedFunction(closure)
				
			elseif typeof(closure) == "Instance" then
				local cloned = env.cloneref(closure)
				env.protectfunction(cloned)
				dest[name] = cloned
				
			else
				dest[name] = closure
			end
		end
		
		return dest
	end

		for i, v in pairs(x) do

			else
				funct(i, v)
			end
		end

	function protected:GetUndetectedContent(contentId: string, isImage: boolean, options: {any})
		local as: AssetService = self:GetService("AssetService")
		
		if not contentId then
			local succ, editable = pcall(as.CreateEditableImage, as, options)
			if not succ or not editable then
				wave.console.warn("Unable to create undetected content.")
				return Content.none
			end
			
			return Content.fromObject(editable), editable
		end
		
		assetsCache[contentId] = assetsCache[contentId] or
			(isImage and Content.fromObject(as:CreateEditableImageAsync(contentId, options))
				or Content.fromObject(as:CreateEditableMeshAsync(contentId, options)))

		return assetsCache[contentId] or Content.none
	end

	protected.GetInstance = LPH_NO_VIRTUALIZE(function(self, class: string, properties: {any}, ...)
		local instance = self:GCProtect(self:GetReference(Instance).new(class))
		instance.Name = self:IsStudio() and (properties.Id or class) or wave.utils.strings.random(nil)
		properties = self:ProtectTable(properties or {})
		properties.Id = nil

		instance.Parent = properties.Parent
			and wave.utils.instances.getRobloxInstance(properties.Parent)
			or wave.utils.instances.dynamicParent(nil)

		properties.Parent = nil
		
		local customProps = {}
		for i, v in properties do
			if wave.utils.strings.find(i, "_") then
				customProps[i] = v
				
			elseif typeof(instance[i]) == "RBXScriptSignal" then
				instance[i]:Connect(v);
				
			elseif i ~= "Parent" and i ~= "Name" then
				local props = tosecure[i]
				if props then
					v = self:ProtectAsset(v)
					if useUndContent and props ~= true then
						task.spawn(function()
							local f = props
							if typeof(props) ~= "function" then
								f = function(inst: Instance, contentId: string)
									inst[props[1]] = protected:GetUndetectedContent(contentId, props[2])
								end
							end
							
							local s, callback = pcall(f, instance, v)					
							instance:GetPropertyChangedSignal(i):Connect(function()
								local cid = instance[i]
								if cid == "" then return end
									
								instance[i] = ""
								if callback and typeof(callback) == "function" then
									pcall(callback, cid)
								else
									pcall(f, instance, cid)
								end
							end)
							
							--wave.env.rconsoleerr("Failed to load Undetected Content: ".. callback)
						end)
					else
						instance[i] = v
					end
					
					instance:GetPropertyChangedSignal(i):Connect(function()
						if instance[i] == "" then return end
						self:ProtectAsset(instance[i])
					end)
				else
					instance[i] = v
				end
			end
		end

		local children = self:GCProtect({})
		for _, v in ipairs({...}) do
			if typeof(v) == "Instance" then
				v.Parent = instance;
				
			elseif typeof(v) == "function" then
				for _, child in ipairs(v(instance)) do
					wave.utils.tables.insert(children, child)
					
					if typeof(child) == "Instance" then
						child.Parent = instance;
						
					else
						child:SetParent(instance)
					end
				end
				
				continue
			else
				v:SetParent(instance)
			end
			
			wave.utils.tables.insert(children, v)
		end

		return instance, children, customProps
	end)
	
	function protected:ProtectTable(data: {any})
		self:GCProtect(data)
		
		deepForeach(data, self:GCProtect(LPH_NO_VIRTUALIZE(function(key: any, value: any)
			if typeof(key) == "function" or typeof(key) == "Instance" then
				self:GCProtect(key)
			elseif typeof(key) == "table" then
				self:ProtectTable(key)
			end
			
			if typeof(value) == "function" or typeof(value) == "Instance" then
				self:GCProtect(value)
			elseif typeof(value) == "table" then
				self:ProtectTable(value)
			end
		end)))
		
		return data
	end

		parent.Name = wave.utils.strings:GetRandomString()

		for _, instance in ipairs(parent:GetDescendants()) do
			instance.Name = wave.utils.strings:GetRandomString()
		end
	end

--[[ Signal ]]

do
	local baseSignal, signals, cache = {}, {}, wave.protected:GCProtect({})
	local counter = 0

	function baseSignal:Once(funct: (any))
		local conn; conn = self:Connect(function(...)
			conn:Disconnect()
			funct(...)
		end)
		
		return conn
	end

	function baseSignal:Wait()
		local thread = coroutine.running()
		self:Once(function(...)
			task.spawn(thread, ...)
		end)
		
		return coroutine.yield()
	end

	function baseSignal:SetId(id: string)
		if cache[id] then
			error("Signal id already in use.")
		end

		cache[id] = self
		cache[self.id] = nil
		self.id = id

		return self
	end

	function baseSignal:Remove()
		self:Dispose()

		cache[self.id] = nil
	end

	function baseSignal:Call(...: any)
		if self.Invoke then
			return self:Invoke(...)
		end

		self:Fire(...)
	end

	function signals.newEvent(id: string?): signalEvent
		local finalId = (id and not cache[id]) and id or (function()
			counter += 1
			return "Event_" .. counter
		end)()

		local connections, firing, adding, lastArgs = {}, false, false, nil
		local event = wave.utils.tables.concatenate({
			id = finalId

		}, baseSignal)

		function event:Connect(funct: (any)): connection
			local connection = {}

			local wasFiring, exit = false, false
			function connection:Disconnect()
				exit = true
				connections[self] = nil
			end

			adding = true
			task.spawn(LPH_NO_VIRTUALIZE(function()
				while firing do
					wasFiring = true
					task.wait()
				end

				if not exit then
					connections[connection] = funct
					if wasFiring then
						task.spawn(funct, table.unpack(lastArgs))
					end
				end

				adding = false
			end))

			return connection
		end

		function event:Fire(...)
			task.spawn(LPH_NO_VIRTUALIZE(function(...)
				while adding do
					task.wait()
				end

				firing = true
				lastArgs = {...}

				for _, funct in pairs(connections) do
					task.spawn(funct, ...)
				end

				firing = false
				return self
			end), ...)
		end

		function event:Dispose()
			connections = {}
			return self
		end

		if id then
			cache[finalId] = event
		end

		return event
	end

	function signals.newFunction(id: string?): signalFunction
		local finalId = (id and not cache[id]) and id or (function()
			counter += 1
			return "Function_" .. counter
		end)()

		local conn, callback
		local fn = wave.utils.tables.concatenate({
			id = finalId

		}, baseSignal)

		function fn:Connect(funct: (any)): connection
			if conn then
				conn:Disconnect()
			end

			local connection = {}
			function connection:Disconnect()
				connection = nil
				callback = nil
			end

			callback = funct
			conn = connection
			return connection
		end

		function fn:Invoke(...)
			if callback then
				return callback(...)
			end
		end

		function fn:Dispose()
			callback = nil
			return self
		end

		if id then
			cache[finalId] = fn
		end

		return fn
	end

	function signals:Get(id: string): signalEvent|signalFunction
		return cache[id]
	end
	
	wave.protected:ProtectTable({baseSignal, signals})
	wave.signals = signals
end

			function cache:Push(key: string, value: any, ifMissing: boolean)
				if ifMissing and map[key] ~= nil then return map[key] end
				
				map[key] = value
				wave.utils.tables.insert(idxs, key)
				
				while #idxs > limit do
					map[idxs[1]] = nil
					wave.utils.tables.remove(idxs, 1)
				end
				return map[key]
			end
			
			function cache:Get(key: string)
				return map[key]
			end
			
			function cache:GetSize()
				return #idxs
			end
			
			function cache:GetNewest(num: number?)
				num = wave.utils.maths.clamp(num, 1, #idxs)
				local newest = {}
				
				for i=#idxs, #idxs -num +1, -1 do wave.utils.tables.insert(newest, idxs[i]) end
				return newest
			end
			
			function cache:GetOldest(num: number?)
				num = wave.utils.maths.clamp(num, 1, #idxs)
				local oldest = {}

				for i=1, num do wave.utils.tables.insert(oldest, idxs[i]) end
				return oldest
			end
			
			function cache:GetAll()
				return wave.utils.tables.listCopy(idxs)
			end
			
			function cache:SetThreshold(limit: number)
				limit = wave.utils.maths.clamp(limit, 1, math.huge)
				return self
			end
			
			function cache:Remove()
				cacheCache[id] = nil
			end

			cache:SetThreshold(limit or math.huge)
			wave.protected:ProtectTable(cache)
			cacheCache[id] = cache
			return cache
		end
		
		function caches:Get(id: string)
			return cacheCache[id]
		end
		
		wave.protected:ProtectTable(caches)
		utils.caches = caches
	end

			local convertType = wave.protected:GCProtect(function(a, p1, p2, p3, p4) -- Converts data to bigInt if possible
				local t, n = {}, nil
				if type(a) == "number" then
					assert(a == a and a >= 0 and math.abs(a) ~= math.huge, "Unable to cast value to bigInt")
					a = math.floor(a)
					while a ~= 0 do
						table.insert(t, 1, a % 16777216)
						a = math.floor(a / 16777216)
					end
				elseif type(a) == "string" then
					if string.match(a, "^[0_]*$") then
						t[1] = 0
					elseif string.match(a, "^_*0_*[Xx][%x_]+$") or string.match(a, "^_*0_*[Bb][01_]+$") then
						local x = if string.match(a, "[Xx]") then 16 else 2
						n = string.gsub(string.match(a, "0_*.[0_]*(.+)"), "_", "")
						n = string.rep("0", - string.len(n) % if x == 16 then 6 else 24) .. n
						for i in string.gmatch(n, "(......" .. if x == 16 then ")" else "..................)")
						do
							table.insert(t, tonumber(i, x))
						end
					elseif string.match(a, "^_*[%d_]*%.?[%d_]*$") then
						table.clear(p1)
						table.clear(p2)
						p1[1] = 10000000
						p2[1] = 1
						n = string.gsub(string.match(a, "_*[0_]*([%d_]*)%.?.-$"), "_", "")
						n = string.rep("0", - string.len(n) % 7) .. n
						for i in string.gmatch(string.reverse(n), "(.......)") do
							table.clear(p3)
							p3[1] = tonumber(string.reverse(i))
							mul(p3, p2, p4)
							table.clear(p3)
							add(p4, table.move(t, 1, #t, 1, p3), t)
							table.clear(p3)
							mul(table.move(p2, 1, #p2, 1, p3), p1, p2)
						end
					else
						error("Unable to cast value to bigInt")
					end
				elseif type(a) == "table" then
					for i, j in ipairs(a) do
						assert(type(j) == "number" and math.floor(j) == j and 0 <= j and j < 16777216,
							"Unable to cast value to bigInt")
						t[i] = j
					end
					if #t == 0 then
						t[1] = 0
					end
				else
					error("Unable to cast value to bigInt")
				end
				return t
			end)

				end,

					end

						table.insert(r, 1, 0)
					end
					for _ = 1, #r / 3 do
						n = bit32.lshift(r[1], 16) + bit32.lshift(r[2], 8) + r[3]

							table.insert(r, n)
							x = true
						end

						table.remove(r, 1)
						table.remove(r, 1)

		function versioning:Compare(ver: string)
			local this, other = utils.strings.split(current, "."), utils.strings.split(ver, ".")
			local max = utils.maths.max(#this, #other) +1
			local thisScore, otherScore = 0, 0

			for i, v in ipairs(this) do
				this[i] = tonumber(v)
			end

			for i, v in ipairs(other) do
				other[i] = tonumber(v)
			end

			for i=1, max -1 do
				if this[i] and other[i] then
					if this[i] > other[i] then
						thisScore += max -i

					elseif this[i] < other[i] then
						otherScore += max -i

					end
				elseif this[i] then
					if thisScore >= otherScore and this[i] > 0 then
						thisScore += max -i
					end
				else
					if otherScore >= thisScore and other[i] > 0 then
						otherScore += max -i
					end
				end
			end

			if thisScore > otherScore then
				return wave.enums.versioning.older
			elseif thisScore < otherScore then
				return wave.enums.versioning.newer
			else
				return wave.enums.versioning.match
			end
		end

		utils.versioning = versioning

					elseif operators[ch] then
						flush()
						insert(tokens, ch)
						i = i + 1

						-- Identifier/number
					elseif ch:match("[%w_]") then
						buf[#buf+1] = ch
						i = i + 1

						-- Anything else (whitespace, etc.)
					else
						flush()
						insert(tokens, ch)
						i = i + 1
					end

				elseif state == "comment" then
					buf[#buf+1] = ch
					i = i + 1
					if ch == "\n" then
						flush()
						state = "normal"
					end

				elseif state == "long_comment" then
					buf[#buf+1] = ch
					-- detect closing ]] 
					if sub(source, i-1, i) == long_delim then
						flush()
						state = "normal"
					end
					i = i + 1

				elseif state == "string" then
					buf[#buf+1] = ch
					i = i + 1
					-- end on unescaped delim or on newline
					if (ch == delim and sub(source, i-2, i-2) ~= "\\") or ch == "\n" then
						flush()
						state = "normal"
					end

				elseif state == "long_string" then
					buf[#buf+1] = ch
					-- detect closing ]]
					if sub(source, i-1, i) == long_delim then
						flush()
						state = "normal"
					end
					i = i + 1

				end
				
				if i%10000 == 0 then task.wait() end
			end

			flush()
			return tokens
		end)

		-- apply getHighlight() to each token
		highlighter.run = LPH_NO_VIRTUALIZE(function(source, maxc)
			local tokens = highlighter.tokenize(source, maxc)
			for idx, tok in ipairs(tokens) do
				local hl = getHighlight(tokens, idx)
				if hl then
					-- escape < and >
					local esc = tok:gsub("<","&lt;"):gsub(">","&gt;")
					tokens[idx] = string.format(
						'<font color="#%s">%s</font>',
						hl:ToHex(),
						esc
					)
				end
			end
			return table.concat(tokens)
		end)

		highlighter.setTokenColors()
		utils.highlighter = wave.protected:ProtectTable(highlighter)
	end

		function inputs.makeDraggable(instance)
			if inputs.isDraggable(instance) then return end

			local dragging, previousMousePos = false, nil
			local object: GuiObject = instance.instance
			local initialMousePos, startPos = nil, nil

			object.Active = true
			local Update = wave.protected:GCProtect(LPH_NO_VIRTUALIZE(function(dt)
				if not dragging or isDraggingSliders then return end
				if not object or not object.Active or not object.Visible then return end

				local currentMousePos = USER_INPUT_SERVICE:GetMouseLocation()
				if previousMousePos and (currentMousePos -previousMousePos).Magnitude > 100 then
					previousMousePos = nil
					dragging = false
					return
				end
				
				previousMousePos = currentMousePos
				local delta = currentMousePos -initialMousePos
				local xGoal = startPos.X.Offset +delta.X
				local yGoal = startPos.Y.Offset +delta.Y
				
				local viewportSize = object.Parent.AbsoluteSize
				local objectSize = object.AbsoluteSize
				local anchor = object.AnchorPoint
				
				local xOffsize = objectSize.X -(objectSize.X *anchor.X) -viewportSize.Y *.1
				local yOffsize = objectSize.Y -(objectSize.Y *anchor.Y) -viewportSize.Y *.1

				xGoal = utils.maths.clamp(xGoal, -xOffsize, viewportSize.X +xOffsize)
				yGoal = utils.maths.clamp(yGoal, -yOffsize, viewportSize.Y +yOffsize)
				object.Position = UDim2.fromOffset(xGoal, yGoal)
			end))
			
			local inputConn = object.InputBegan:Connect(wave.protected:GCProtect(LPH_NO_VIRTUALIZE(function(input)
				if not dragging and not thumbstick and (input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch) then
					dragging = true
					initialMousePos = USER_INPUT_SERVICE:GetMouseLocation()

					local pos = object.Position
					local viewportSize = object.Parent.AbsoluteSize
					startPos = UDim2.fromOffset(pos.X.Scale * viewportSize.X + pos.X.Offset, pos.Y.Scale * viewportSize.Y + pos.Y.Offset)
					object.Position = startPos

					input.Changed:Connect(function()
						if input.UserInputState == Enum.UserInputState.End then dragging = false end
					end)
				end
			end)))
			
			wave.renderer:BindToRenderer(`{instance.interface.id}-{instance.id}`, wave.renderer.FrameworkPriority, Update):Render()
			dragCache[instance] = {inputConn}
		end

				conn2 = nil
				conn = nil
			end

			conn = btn.MouseButton1Down:Connect(LPH_NO_VIRTUALIZE(function()
				if not input.enabled then
					return
				end

				lastPos = btn.AbsolutePosition
				lastPress = os.clock()

				if not waitForRelease then
					tasks.spawn(function()
						local exit = false

						local callback
						callback = fetch:Connect(function()
							if not exit then
								input.OnShortPress:Fire(os.clock() - lastPress, btn)
								exit = true
							end
						end)

						tasks.wait(setts.long_click + 0.05)

						if not exit then
							exit = true
							if inputs.isDraggable(button) and inputs.hasBeenDragged(lastPos, btn.AbsolutePosition) then
								return
							end

							input.OnLongPress:Fire(os.clock() - lastPress, btn)
						end
					end)
				end
			end))

			conn2 = btn.MouseButton1Up:Connect(LPH_NO_VIRTUALIZE(function()
				if not input.enabled or (inputs.isDraggable(button) and inputs.hasBeenDragged(lastPos, btn.AbsolutePosition)) then
					return
				end

				if not waitForRelease then
					fetch:Fire()
					return
				end

				local clickTime = os.clock() - lastPress
				if lastPress and clickTime >= setts.long_click then
					input.OnLongPress:Fire(clickTime, button)
				else
					input.OnShortPress:Fire(clickTime, button)
				end
			end))

			wave.protected:ProtectTable(input)
			buttonsCache[id] = input
			return input
		end

						exit = true
					end)

					end

--[[ Storage ]]

do
	local storage, directories = {}, wave.protected:GCProtect({})
	local function fullPath(directory: string, name: string)
		directory = directories[directory]
		return directory and (name and (directory.path .. "/" .. name) or directory.path) or name
	end

	function storage:CreateDirectory(name: string, path: string, parent: directory|string): directory
		if directories[name] then
			wave.console.error("Directory name already in use.")
		end
		
		local lastPath = path
		if parent then
			if typeof(parent) == "table" then
				path = parent.path .. "/" .. path
			else
				local dir = directories[parent]
				if dir then
					path = dir.path .. "/" .. path
				end
			end
		end
		
		-- Protects folder if it becomes a Custom Asset
		wave.protected:ProtectAsset(name)
		wave.protected:ProtectAsset(`{name}/`)
		
		wave.protected:ProtectAsset(path)
		wave.protected:ProtectAsset(`{path}/`)

		if not wave.env.isfolder(path) then
			wave.env.makefolder(path)
		end

		local directory = {
			lastPath = lastPath,
			parent = parent,
			name = name,
			path = path
		}

		function directory:AddSubDirectory(name: string, path: string)
			return storage:CreateDirectory(name, path, self)
		end

		function directory:Remove()
			if self.parent then
				if typeof(parent) == "table" then
					self.parent:Remove()
				else
					local dir = directories[parent]
					if dir then
						dir:Remove()
					end
				end
			end

			directories[self.name] = nil
		end
		
		function directory:WriteFile(name: string, source: string, encryptionKey: string)
			return storage.writeFile(self.name, name, source, encryptionKey)
		end
		
		function directory:ReadFile(name: string, decryptionKey: string)
			return storage.readFile(self.name, name, decryptionKey)
		end
		
		function directory:WriteJsonFile(name: string, source: string, encryptionKey: string)	
			return storage.writeJsonFile(self.name, name, source, encryptionKey)
		end
		
		function directory:ReadJsonFile(name: string, decryptionKey: string)
			return storage.readJsonFile(self.name, name, decryptionKey)
		end
		
		function directory:DeleteFile(name: string)
			return storage.deleteFile(self.name, name)
		end
		
		function directory:IsFolder(name: string)
			return storage.isFolder(self.name, name)
		end
		
		function directory:MakeFolder(name: string)
			return storage.makeFolder(self.name, name)
		end
		
		function directory:DeleteFolder(name: string)
			return storage.deleteFolder(self.name, name)
		end
		
		function directory:IsFile(name: string)
			return storage.isFile(self.name, name)
		end
			
		function directory:ListFiles(path: string)
			return storage.listFiles(self.name, path)
		end
		
		wave.protected:ProtectTable(directory)
		directories[name] = directory
		return directory
	end

	function storage:RemoveDirectory(name): directory
		local dir = directories[name]

		if not dir or not wave.env.isfolder(dir.path) then
			return false
		end

		wave.env.delfolder(dir.path)
		return true
	end

	function storage:GetDirectory(name: string): directory
		return directories[name]
	end

	function storage.writeFile(directory: string, name: string, source: string, encryptionKey: string)
		name = fullPath(directory, name)

		-- Protects the file if it becomes a Custom Asset
		wave.protected:ProtectAsset(name)
		wave.protected:ProtectAsset(`{name}/`)

		if encryptionKey then
			source = wave.utils.crypt.base64.encode(wave.utils.crypt.bytes(source, encryptionKey))
		end

		wave.env.writefile(name, source or "")
	end

	function storage.readFile(directory: string, name: string, decryptionKey: string)
		if not storage.isFile(directory, name) then
			return false
		end

		local source = wave.env.readfile(fullPath(directory, name))

		if decryptionKey then
			source = wave.utils.crypt.bytes(wave.utils.crypt.base64.decode(source), decryptionKey)
		end

		return source
	end

	function storage.writeJsonFile(directory: string, name: string, source: string, encryptionKey: string)
		return storage.writeFile(directory, name, wave.utils.http.json.encode(source), encryptionKey)
	end

	function storage.readJsonFile(directory: string, name: string, decryptionKey: string)
		local source = storage.readFile(directory, name, decryptionKey)

		if source then
			return wave.utils.http.json.decode(source)
		end

		return false
	end

	function storage.deleteFile(directory: string, name: string)
		if not storage.isFile(directory, name) then
			return false
		end
		
		wave.env.delfile(fullPath(directory, name))
		return true
	end
	
	function storage.isFolder(directory: string, name: string)
		name = fullPath(directory, name)

		local exists = wave.env.isfolder(name)
		if not exists or exists == 404 then
			return false
		end

		return true
	end
	
	function storage.deleteFolder(directory: string, name: string)
		if not storage.isFolder(directory, name) then
			return false
		end

		wave.env.delfolder(fullPath(directory, name))
		return true
	end
	
	function storage.makeFolder(directory: string, name: string)
		if not storage.isFolder(directory, name) then
			wave.env.makefolder(fullPath(directory, name))
			return true
		end
		
		return false
	end
	
	function storage.isFile(directory: string, name: string)
		name = fullPath(directory, name)
		
		local exists = wave.env.isfile(name)
		if not exists or exists == 404 then
			return false
		end
		
		return true
	end
	
	function storage.listFiles(directory: string, path: string)
		directory = fullPath(directory, path)
		
		local list = wave.env.listfiles(directory)
		if not list or list == 404 then
			return {}
		end
		
		return list
	end

	wave.protected:ProtectTable(storage)
	wave.storage = storage
end

		local success, file = pcall(function()

		if not success then
			file = {}
		end

		curSetts = wave.protected:GCProtect(wave.utils.tables.deepOverwrite(wave.utils.tables.deepCopy(defSetts), file))
		for key, value in pairs(curSetts) do
			changed(key, value, defSetts[key])

		return self
	end

		return curSetts[name]
	end

	function setts:GetDefaultSetting(name: string)
		return defSetts[name]

		if typeof(value) ~= "table" and curSetts[name] == value then
			return self

		curSetts[name] = value
		changed(name, value, defSetts[key])

		if self.autoSave then
			tasks.spawn(self.SaveSettings, self)
		end

		return self
	end

		curSetts = wave.utils.tables.deepCopy(defSetts)

		return signals[name]
	end

--[[ Execution ]]

do
	local execution = {}

	execution.OnExecuted = wave.signals.newEvent()
	execution.OnFileExecuted = wave.signals.newEvent()
	execution.OnSourceExecuted = wave.signals.newEvent()

	function execution:Execute(source: string, sync: boolean, ...: any)
		if sync then
			local suc, res = pcall(wave.env.loadstring, source)
			if suc then
				res(...)
			end
		else
			pcall(wave.env.executecode, source)
		end

		self.OnExecuted:Fire()
		self.OnSourceExecuted:Fire(source)

		return self
	end

	function execution:ExecuteFile(path: string, sync: boolean)
		if not wave.env.isfile(path) then
			return false
		end

		if sync then
			pcall(wave.env.loadfile, path)
		else
			pcall(wave.env.dofile, path)
		end
		
		self.OnExecuted:Fire()
		self.OnFileExecuted:Fire(path)
		
		return self
	end
	
	function execution:ExecuteUrl(url: string, sync: boolean, method: string, headers: {any}, body: any)
		local res = wave.utils.http:Request(url, method, headers, body)
		if res then
			execution:Execute(res.Body, sync)
		end
		
		return self
	end

	wave.protected:ProtectTable(execution)
	wave.execution = execution
end

--[[ Renderer ]]

do
	local RUN_SERVICE: RunService = wave.protected:GetService("RunService")
	local renderer, bindings = {}, wave.protected:GCProtect({})
	renderer.FrameworkPriority = -999

	local function conformPriority(priority: number)
		return priority <= 0 and 1 or priority
	end

	local function unbind(id: string)
		RUN_SERVICE:UnbindFromRenderStep(id)
	end

	function renderer:BindToRenderer(id: string, priority: number?, funct: (any) -> (), ...: any): binding
		if bindings[id] then
			wave.console.error("Id: " .. id .. " already in use in rendering.")
		end

		local args = wave.protected:GCProtect({...})
		local stateChanged = wave.signals.newEvent()
		
		local binding = {
			id = id,
			priority = priority or renderer.FrameworkPriority +1,
			enabled = false
		}
		
		binding.OnStateChanged = stateChanged

		function binding:Pause()
			unbind(self.id)
			if self.enabled then
				self.enabled = false
				stateChanged:Fire(false)
			end
			
			return self
		end

		function binding:Resume()
			unbind(self.id)
			RUN_SERVICE:BindToRenderStep(self.id, self.priority, function(delta: number)
				funct(delta, wave.utils.tables.unpack(args))
			end)

			if not self.enabled then
				self.enabled = true
				stateChanged:Fire(true)
			end
			
			return self
		end

		function binding:Render()
			return binding:Resume()
		end

		function binding:SetPriority(priority: number)
			self.priority = conformPriority(priority)
			if self.enabled then
				self:Resume()
			end

			return self
		end		

		function binding:Unbind()
			unbind(self.id)
			bindings[self.id] = nil
			stateChanged:Fire(false)
		end

		wave.protected:ProtectTable(binding)
		bindings[id] = binding
		return binding
	end

	function renderer:GetOnRendering(): {binding}
		local rendering = {}
		for id, binding in pairs(bindings) do
			if binding.enabled then
				rendering[id] = binding
			end
		end

		return rendering
	end

	function renderer:GetBindings(): {binding}
		local binds = {}
		for id, binding in pairs(bindings) do
			binds[id] = binding
		end

		return binds
	end

	function renderer:GetBinding(id: string): binding
		return bindings[id]
	end

	function renderer:Dispose()
		for id, _ in pairs(bindings) do
			unbind(id)
		end

		bindings = nil
		return self
	end

	wave.protected:ProtectTable(renderer)
	wave.renderer = renderer
end

do -- animator
	local this, configs = {}, wave.protected:GCProtect({ -- Riky47#3355 Dynamic Srpites Animator
		rawin = {"ResX", "ResY", "Duration", "Density"},
		rawout = {"efc", "dur", "resy", "resx"},
		md = 1024, -- Max dimension
		ddur = 1,
		dden = 1
	})

	--this.active = true
	local onLoaded = wave.signals.newEvent()
	this.OnAnimationLoaded = onLoaded
	
	function this.new(id: string, options: {any}, frameGen: (writePixel: (pixelIdx: number, r: number, g: number, b: number, a: number) -> void,
		resolution: Vector2, frameIdx: number, totalFrames: number) -> void, priority: number, ...: any)

		local data = {
			res = options.Resolution or Vector2.new(options.ResX, options.ResY), -- resolution
			resx = options.ResX or options.Resolution.X,
			resy = options.ResY or options.Resolution.Y,
			dur = options.Duration or configs.ddur, -- duration
			den = options.Density or configs.dden, -- frames density
			imgs = options.Outputs or {}, -- image object
			once = false, -- renders once
			elapsed = 0, -- elapsed
			ogimg = {}, -- images from labels
			ogtrn = {}, -- imageTransparencies
			ares = 0, -- all resolution
			lfi = -1, -- last frame index
			efc = 0, -- effective frames count
			fpr = 1, -- frames per row
			ui = 0 -- Upper index
		}

		for _, img in ipairs(data.imgs) do
			wave.utils.tables.insert(data.ogimg, img.Image)
			wave.utils.tables.insert(data.ogtrn, img.ImageTransparency)
		end

		local compute = wave.protected:GCProtect(LPH_NO_VIRTUALIZE(function()
			data.fpr = wave.utils.maths.floor(configs.md /data.resx)
			data.efc = data.fpr *wave.utils.maths.floor(configs.md /data.resy)
			data.efc = wave.utils.maths.min(data.den, data.efc)
			data.ui = data.efc -1

			if data.den > data.efc then
				wave.console.warn(`Clamped frames to {data.efc}`)
			end
		end))

		local content, edit
		local setImgs = wave.protected:GCProtect(function(reset: boolean)
			for i, img in ipairs(data.imgs) do
				if reset then img.Image = data.ogimg[i] else
					img.ImageContent = content or Content.none
				end

				img.ImageRectSize = reset and Vector2.zero or data.res
				img.ImageTransparency = data.ogtrn[i]
				img.ImageRectOffset = Vector2.zero
				data.elapsed = 0
			end
		end)

		local binding
		local sprite = wave.protected:GCProtect(LPH_NO_VIRTUALIZE(function()
			if edit then edit:Destroy() end
			
			local totalRows = wave.utils.maths.ceil(data.efc /data.fpr)
			data.ares = Vector2.new(data.fpr *data.resx, totalRows *data.resy)
			content, edit = wave.protected:GetUndetectedContent(nil, true, {
				Size = data.ares
			})
			
			if not edit then
				wave.console.warn("Unable to use undetected content.")
			end

			setImgs(false)
		end))

		local resume = wave.protected:GCProtect(function()
			for i, img: ImageLabel in ipairs(data.imgs) do
				img.ImageTransparency = data.ogtrn[i]
			end

			setImgs(false)
			binding:Resume()
		end)

		local setFrame = wave.protected:GCProtect(LPH_NO_VIRTUALIZE(function(frameIndex: number)
			local col, row = frameIndex %data.fpr, wave.utils.maths.floor(frameIndex /data.fpr)
			local off = Vector2.new(col *data.resx, row *data.resy)
			for _, img: ImageLabel in ipairs(data.imgs) do
				img.ImageRectOffset = off
			end
		end))

		compute()
		binding = wave.renderer:BindToRenderer(id, priority or wave.renderer.FrameworkPriority +1, LPH_NO_VIRTUALIZE(function(delta)
			--if not this.active then return end
			data.elapsed = (data.elapsed +delta) %data.dur
			local rawIndex = (data.elapsed /data.dur) *data.efc

			local frameIndex = wave.utils.maths.floor(rawIndex)
			if data.lfi == frameIndex then return end

			data.lfi = frameIndex
			if frameIndex >= data.ui then
				if data.once then binding:Stop() end
				frameIndex = data.ui
			end

			setFrame(frameIndex)
		end), ...)

		if options.Raw then
			binding.Loaded = pcall(sprite) and pcall(edit.WritePixelsBuffer, edit, Vector2.zero, data.ares, options.Raw)
			if not binding.Loaded then
				for key, _ in pairs(options) do
					local id = `og_{key}`
					if options[id] then
						options[key] = options[id]
						options[id] = nil
					end
				end

				data.res = options.Resolution
				if not data.res then wave.console.error("No default resolution specified") end

				data.dur = options.Duration or configs.ddur
				data.den = options.Density or configs.dden
				data.resx, data.resy = data.res.X, data.res.Y
				compute()
			end
		else
			binding.Loaded = false
		end

		binding.Generate = LPH_NO_VIRTUALIZE(function(self, resolution: Vector2?, density: number?)
			self:Pause()
			if resolution and data.res ~= resolution then
				data.res = resolution; data.resx = resolution.X; data.resy = resolution.Y
				if density then data.den = density end
				compute(); sprite()

			elseif not edit then sprite() end			
			if not edit then return end

			for idx=0, data.ui, 1 do
				local col, row = idx %data.fpr, wave.utils.maths.floor(idx /data.fpr)
				local buff = buffer.create(data.resx * data.resy *4)

				local function writePixel(idx: number, r: number, g: number, b: number, a: number)
					buffer.writeu8(buff, idx, r); buffer.writeu8(buff, idx +1, g); buffer.writeu8(buff, idx +2, b); buffer.writeu8(buff, idx +3, a)
				end

				frameGen(writePixel, data.res, idx, data.efc)
				edit:WritePixelsBuffer(Vector2.new(col *data.resx, row *data.resy), data.res, buff)
			end

			binding.Loaded = true
			onLoaded:Fire(binding)
			return self
		end)

		function binding:SetDuration(duration: number)
			data.dur = duration
			return self
		end

		function binding:Render() -- Overwrite
			if not self.Loaded then return self end
			data.once = false
			data.elapsed = 0
			resume()

			return self
		end

		function binding:RenderOnce(callback: () -> ())
			if not self.Loaded then return self end
			data.elapsed = 0
			data.once = true

			local conn
			conn = self.OnStateChanged:Connect(function(state: boolean)
				if not state then 
					if callback then callback() end
					conn:Disconnect()
				end
			end)

			resume()
			return self
		end

		function binding:Stop()
			self:Pause()
			for _, img: ImageLabel in ipairs(data.imgs) do
				img.ImageTransparency = 1
			end
		end

		function binding:SetEnabled(state: boolean)
			if not state then self:Pause() end
			setImgs(not state)
			return self
		end

		function binding:SetFrame(n: number)
			if data.efc < 1 then return end
			setFrame(wave.utils.maths.clamp(n, 1, data.efc))
			return self
		end

		function binding:GetFrameCount()
			return data.efc
		end

		function binding:GetMaxFPS()
			return data.efc /data.dur
		end

		function binding:GetWeight()
			return buffer.len(edit:ReadPixelsBuffer(Vector2.zero, data.ares))/1024/1024
		end

		binding.GetRaw = LPH_NO_VIRTUALIZE(function(self, raw: string)
			if not edit then return false end

			local sprite = edit:ReadPixelsBuffer(Vector2.zero, data.ares)
			local slen = buffer.len(sprite)

			local buff = buffer.create(slen +#configs.rawout *2)
			buffer.copy(buff, 0, sprite, 0, slen)

			for idx, key in ipairs(configs.rawout) do
				buffer.writeu16(buff, slen +(idx -1) *2, wave.utils.maths.floor(data[key] *10))
			end

			return buffer.tostring(buff)
		end)
		
		if binding.Loaded then onLoaded:Fire(binding) end
		return binding
	end

	this.fromRaw = LPH_NO_VIRTUALIZE(function(id: string, options: {any}, raw: string, frameGen: (idx: number, frames: number) -> buffer, priority: number, ...: any)
		if not options.Resolution then wave.console.error("No default resolution specified") end
		local buff = buffer.fromstring(raw or "")
		local len = buffer.len(buff)

		local plen = #configs.rawin *2
		if len > plen then
			for idx, key in ipairs(configs.rawin) do
				local value = buffer.readu16(buff, len -idx *2)
				options[`og_{key}`] = options[key]
				options[key] = value /10
			end

			if Vector2.new(options.ResX, options.ResY) == options.Resolution then
				local slen = len -plen
				local sprite = buffer.create(slen)
				buffer.copy(sprite, 0, buff, 0, slen)
				options.Raw = sprite
			end
		end

		return this.new(id, options, frameGen, priority, ...)
	end)

	this.computePixels = LPH_NO_VIRTUALIZE(function(resolution: Vector2, onPixel: (idx: number, x: number, y: number) -> void)
		for y = 0, resolution.Y -1 do
			for x = 0, resolution.X -1 do
				onPixel((y *resolution.X +x) *4, x, y)
			end
		end
	end)

	function this:Get(id: string)
		return wave.renderer:GetBinding(id)
	end
	
	wave.animator = wave.protected:ProtectTable(this)
end

--[[ Interface ]]

do
	local interface, interfaces = {}, wave.protected:GCProtect({})
	local interfaceCounter = 0

	function interface.new(interfaceId: string): interface
		interfaceId = interfaceId or (function()
			interfaceCounter += 1
			return "UI_" .. interfaceCounter
		end)()

		local gui = wave.protected:GetInstance("ScreenGui", {
			DisplayOrder = 1999999999,
			IgnoreGuiInset = true,
			ResetOnSpawn = false,
			Enabled = false
		})

		local instances, components, baseComponents, imaps = wave.protected:GCProtect({}), wave.protected:GCProtect({}), wave.protected:GCProtect({}), wave.protected:GCProtect({})
		local instanceCounter, componentCounter = 0, 0
		local instanceAdded = wave.signals.newEvent()
			
		local newInterface = {
			instance = gui,
			id = interfaceId
		}
		
		local collections = wave.protected:GCProtect({})
		local addToCollection = wave.protected:GCProtect(LPH_NO_VIRTUALIZE(function(tag: string, instance: instance)
			if not collections[tag] then collections[tag] = wave.protected:GCProtect({}) end
			wave.utils.tables.insert(collections[tag], instance)
		end))
		
		local remFromCollection = wave.protected:GCProtect(LPH_NO_VIRTUALIZE(function(tag: string, instance: instance)
			if not collections[tag] then return end
			local found = wave.utils.tables.find(collections[tag], instance)
			if found then wave.utils.tables.remove(collections[tag], found) end
			if #collections[tag] < 1 then collections[tag] = nil end
		end))
		
		local wipeFromCollection = wave.protected:GCProtect(LPH_NO_VIRTUALIZE(function(instance: instance)
			for tag, _ in pairs(collections) do remFromCollection(tag, instance) end
		end))
		
		newInterface.OnInstanceAdded = instanceAdded
		function newInterface:SetEnabled(enabled: boolean)
			self.instance.Enabled = enabled
			return self
		end

		function newInterface:Remove()
			self.instance:Destroy()

			--instances = nil
			--interface = nil
			interfaces[interfaceId] = nil
		end
		
		function newInterface:GetByTag(tag: string)
			return collections[tag] or {}
		end

		function newInterface:AddInstance(class: string, properties: {any}, ...: any): instance
			properties = properties or {}
			properties.Parent = properties.Parent or self.instance
			local instanceId = properties.Id or (function()
				instanceCounter += 1
				return "Instance_" .. instanceCounter
			end)()
			
			local ogtags, tags = properties.Tags, wave.protected:GCProtect({})
			properties.Tags = nil
			
			local onRemove = properties.OnRemoved
			properties.OnRemoved = nil

			local instance, children, customProps = wave.protected:GetInstance(class, properties, ...)
			local customParent = nil
			local object = {
				interface = newInterface,
				instance = instance,
				id = instanceId,
			}
			
			wave.utils.tables.concatenate(object, customProps)
			local addtag = wave.protected:GCProtect(function(tag: string)
				wave.utils.tables.insert(tags, tag)
				addToCollection(tag, object)
			end)
			
			local remtag = wave.protected:GCProtect(function(tag: string)
				remFromCollection(tag, object)
				local found = wave.utils.tables.find(tags, tag)
				if found then wave.utils.tables.remove(tags, found) end
			end)
			
			function object:AddTags(tag: string|{string})
				if typeof(tag) == "string" then
					addtag(tag)
				else
					for _, v in ipairs(tag) do addtag(v) end
				end
				return self
			end
			
			function object:GetTags()
				return wave.utils.tables.deepCopy(tags)
			end
			
			function object:HasTags(tag: string|{string})
				if typeof(tag) == "string" then
					return wave.utils.tables.find(tags, tag) and true or false
				else
					for _, v in ipairs(tag) do
						if not wave.utils.tables.find(tags, v) then return false end
					end
					return true
				end
			end
			
			function object:RemoveTags(tag: string|{string})
				if typeof(tag) == "string" then
					remtag(tag)
				else
					for _, v in ipairs(tag) do
						remtag(v)
					end
				end
				
				return self
			end

			function object:SetId(id: string)
				if instances[id] then
					wave.console.error("Instance id already in use.")
				end
				
				instances[id] = self
				instances[self.id] = nil
				self.id = id
				return self
			end

			function object:SetParent(parent: instance|Instance)
				if typeof(parent) ~= "Instance" then
					customParent = parent
				end

				self.instance.Parent = (typeof(parent) == "Instance" and parent or parent.instance) or wave.utils.instances.dynamicParent()
				return self
			end
			
			function object:GetParent()
				return customParent or {instance = self.instance.Parent}
			end
			
			function object:Tween(tweenInfo: TweenInfo, propertyTable: {any})
				local tween = wave.protected:GetService("TweenService"):Create(self.instance, tweenInfo, propertyTable)
				tween:Play()
				
				return tween
			end

			function object:SetDraggable(draggable: boolean)
				if draggable then
					wave.utils.inputs.makeDraggable(self)
				else
					wave.utils.inputs.makeUndraggable(self)
				end

				return self
			end

			function object:Remove()
				instances[instanceId] = nil
				imaps[instance] = nil
				
				if #tags > 0 then wipeFromCollection(object) end
				self.instance:Destroy()
				if onRemove then onRemove() end
			end
			
			for _, child in ipairs(wave.utils.tables.concatenate({...}, children)) do
				if typeof(child) ~= "Instance" then
					child:SetParent(object)
				end
			end

			instance.AncestryChanged:Connect(function(child, parent)
				if not parent and instances[instanceId] then
					instances[instanceId] = nil
					imaps[instance] = nil
					
					if #tags > 0 then wipeFromCollection(object) end
					object.instance = nil
					if onRemove then onRemove() end
				end
			end)
			
			wave.protected:ProtectTable(object)
			instances[instanceId] = object
			imaps[instance] = object
			
			if ogtags then
				object:AddTags(ogtags)
				ogtags = nil
			end

			instanceAdded:Fire(object)
			return object
		end

		function newInterface:CreateComponent(name: string): baseComponent
			if baseComponents[name] then
				wave.console.error("Component already exists")
			end

			local newComponent = {}
			newComponent.OnCreating = wave.signals.newFunction()

			function newComponent:Remove()
				self.OnCreating:Remove()
				baseComponents[name] = nil
			end

			baseComponents[name] = newComponent
			return newComponent
		end

		function newInterface:AddComponent(class: string, properties: {any}, ...): component
			properties = properties or {}
			properties.Parent = properties.Parent or self.instance

			--local customParent = nil
			local id = properties.Id or (function()
				componentCounter += 1
				return "Component_" .. componentCounter
			end)()

			local component = {}
			local newInstance = baseComponents[class].OnCreating:Invoke(properties, ...)
			if not newInstance or not newInstance.instance then
				wave.console.error("No instance returned for component '" .. class .. "'")
			end

			for k, v in pairs(newInstance) do component[k] = v end
			component.id = id
			
			function component:SetId(id: string)
				if components[id] then
					wave.console.error("Component id already in use.")
				end
				
				components[id] = self
				components[self.id] = nil
				self.id = id
				return self
			end

			--[[
			function component:SetParent(parent: instance|Instance)
				if typeof(parent) ~= "Instance" then
					customParent = parent
				end

				self.instance.Parent = (typeof(parent) == "Instance" and parent or parent.instance) or wave.utils.instances.dynamicParent()
				return self
			end

			function component:GetParent()
				return customParent or {instance = self.instance.Parent}
			end

			function component:Tween(tweenInfo: TweenInfo, propertyTable: {any})
				local tween = wave.protected:GetService("TweenService"):Create(self.instance, tweenInfo, propertyTable)
				tween:Play()
				
				return tween
			end

			function component:SetDraggable(draggable: boolean)
				if draggable then
					wave.utils.inputs.makeDraggable(self)
				else
					wave.utils.inputs.makeUndraggable(self)
				end

				return self
			end]]

			function component:Remove()
				newInstance:Remove()
				self.instance = nil
			end

			if not component.instance.Parent then
				component.instance.Parent = properties.Parent or wave.utils.instances.dynamicParent()
			end

			components[id] = component
			return component
		end

		function newInterface:Get(id: string|Instance): instance|component|baseComponent
			return instances[id] or components[id] or baseComponents[id] or imaps[id]
		end

		wave.protected:ProtectTable(newInterface)
		interfaces[interfaceId] = newInterface
		return newInterface
	end

	function interface:Get(id: string): interface
		return interfaces[id]
	end

	wave.protected:ProtectTable(interface)
	wave.interface = interface
end

wave.protected:ProtectTable(wave)
